# Pitch pin Implementation Plan

> **For agentic workers:** Jonno's standing instruction: execute this inline (superpowers:executing-plans), not with per-task subagents. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first draft of Pitch pin: a daily board of six goals where the player pins where each was struck from, scored in metres or points against a lazy-pin par.

**Architecture:** A self-contained game in `games/pin/` (the Rondo pattern) with thin Next.js route files under `app/`. Pure modules (`pitch`, `score`, `line`, `boards`, `week`, `progress`) carry all the rules and are unit-tested. Goal coordinates live only on the server (`games/pin/server.ts` reading committed JSON); the page ships goals without them and `POST /api/pin/throw` returns one answer per thrown pin. Two scripts build the content: StatsBomb classics (hand-picked) and Understat recent goals with the week's boards.

**Tech Stack:** Next.js 15 (App Router), React 19, TypeScript strict, Vitest, tsx for scripts. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-08-pitch-pin-design.md`

**Two simplifications against the spec** (recorded in the spec in Task 15):
- Daily boards are one file, `games/pin/data/days.json` (date → six slugs), not a folder of files. A static JSON import needs no file-tracing config and no `readdir`.
- The components are three files, not five: `Pitch.tsx` (the SVG), `PitchPin.tsx` (board state, goal card, reveal panel) and `FullTime.tsx`.

**Facts verified on real data before writing this plan (2026-10-08):**
- All 44 classics below resolve to exactly one non-penalty StatsBomb goal by match id, player and minute.
- Understat shot fields: `id, minute, result, X, Y, xG, player, h_a, situation, shotType, lastAction, h_team, a_team, date` (all strings). League fixtures: `id, isResult, datetime, h.title, a.title, goals.h, goals.a`.
- Rebuilding the score from Understat shots matched the final score in all 50 Premier League 2026/27 matches played so far, including 7 own goals: an `OwnGoal` shot is filed under the side of the player who scored it and counts for the other side.
- Understat is slow (one request took 20 s); the script must time out and retry.
- Understat's latest results are 18–20 Sep 2026; no league fixtures from 21 Sep to 9 Oct; next round 10 Oct.

```ts
/** Day 1 on the epoch, counting up. Before the epoch is day 0 or below. Pitch pin passes its own epoch. */
export function dayNumber(date: string, epoch = EPOCH): number {
  return utcDay(date) - utcDay(epoch) + 1;
}

export function dateOfDay(day: number, epoch = EPOCH): string {
  const ms = (utcDay(epoch) + day - 1) * 86400000;
  return new Date(ms).toISOString().slice(0, 10);
}
```

```ts
/**
 * Pitch pin's geometry.
 *
 * Metres on a nominal 105 × 68 m pitch, attacking half only, drawn with the
 * goal at the top: x across from the attacker's left touchline (0 to 68),
 * y out from the goal line (0 to 52.5). Both data sources stretch every
 * pitch to one size, so these are nominal metres.
 */
export interface Spot { x: number; y: number }

export const WIDTH = 68;
export const HALF = 52.5;
/** The lazy pin: the penalty spot. Par is what it would score. */
export const SPOT: Spot = { x: 34, y: 11 };

/** StatsBomb: 120 × 80, attacking toward x = 120, y = 0 on the attacker's left. */
export function fromStatsBomb(sx: number, sy: number): Spot {
  return { x: (sy * WIDTH) / 80, y: ((120 - sx) * 105) / 120 };
}

/** Understat: 0 to 1, X = 1 at the goal attacked, Y = 0 on the attacker's right (flipped against StatsBomb). */
export function fromUnderstat(X: number, Y: number): Spot {
  return { x: (1 - Y) * WIDTH, y: (1 - X) * 105 };
}

export const distance = (a: Spot, b: Spot): number => Math.hypot(a.x - b.x, a.y - b.y);
export const mirror = (s: Spot): Spot => ({ x: WIDTH - s.x, y: s.y });
export const clamp = (s: Spot): Spot => ({ x: Math.min(WIDTH, Math.max(0, s.x)), y: Math.min(HALF, Math.max(0, s.y)) });
export const inHalf = (s: Spot): boolean => s.x >= 0 && s.x <= WIDTH && s.y >= 0 && s.y <= HALF;
/** Inside the penalty area: 16.5 m out and 20.16 m either side of the middle. */
export const inArea = (s: Spot): boolean => s.y <= 16.5 && Math.abs(s.x - WIDTH / 2) <= 20.16;
/** One decimal place, for what is written to disk. */
export const round1 = (n: number): number => Math.round(n * 10) / 10;

/**
 * The wrong-side note. A goal remembered from a side-on camera often comes
 * back mirrored; when the pin is well off but near the mirror image, say so.
 * Returns the mirrored distance, or null when the note does not apply.
 */
export function mirrorNote(pin: Spot, at: Spot): number | null {
  const d = distance(pin, at);
  const m = distance(pin, mirror(at));
  return d > 5 && m < d / 2 ? m : null;
}
```

```ts
import { distance, SPOT, type Spot } from './pitch';

/**
 * Pitch pin's scoring, both ways, so they can be compared. Distances are
 * stored once; metres and points are two readings of them.
 */
export type Ring = 'bullseye' | 'inner' | 'outer' | 'miss';
export type Mode = 'metres' | 'points';
export interface Tally { metres: number; points: number }

/** The bullseye is 2 m: the two sources disagree by up to 3 m on one goal, so finer would reward noise. */
export function ring(d: number): Ring {
  return d <= 2 ? 'bullseye' : d <= 5 ? 'inner' : d <= 10 ? 'outer' : 'miss';
}

export function points(d: number): number {
  return d <= 2 ? 100 : Math.round(100 * Math.exp(-(d - 2) / 8));
}

export const RING_NAME: Record<Ring, string> = { bullseye: 'Bullseye', inner: 'Inner', outer: 'Outer', miss: 'Miss' };
export const MARK: Record<Ring, string> = { bullseye: '🎯', inner: '🟩', outer: '🟨', miss: '⬜' };

export function tally(ds: readonly number[]): Tally {
  return {
    metres: Math.round(ds.reduce((a, d) => a + d, 0) * 10) / 10,
    points: ds.reduce((a, d) => a + points(d), 0),
  };
}

/** Par: what the lazy pin scores, every goal pinned on the penalty spot. */
export function par(ats: readonly Spot[]): Tally {
  return tally(ats.map((a) => distance(SPOT, a)));
}

/** Metres against par, lower is better: "26 under", "2 over", "level". */
export function againstPar(mine: number, parMetres: number): string {
  const diff = Math.round(parMetres - mine);
  return diff === 0 ? 'level' : diff > 0 ? `${diff} under` : `${-diff} over`;
}

/** The share text: the board, the score against par, one mark a goal. No names, no positions. */
export function shareText(title: string, ds: readonly number[], parT: Tally, mode: Mode): string {
  const t = tally(ds);
  const score = mode === 'points'
    ? `${t.points} / ${ds.length * 100}, par ${parT.points}`
    : `${t.metres.toFixed(1)} m, par ${Math.round(parT.metres)} (${againstPar(t.metres, parT.metres)})`;
  return `Pitch pin ${title}\n${score}\n${ds.map((d) => MARK[ring(d)]).join('')}`;
}
```

```ts
/**
 * The line under a goal. It says what kind of goal it was, never where it
 * was struck from: the position is the answer.
 */
const BANNED = new Set([
  'yard', 'yards', 'metre', 'metres', 'meter', 'meters', 'edge', 'box', 'area', 'post',
  'left', 'right', 'centre', 'center', 'halfway', 'distance', 'range', 'close', 'far',
  'long', 'near', 'angle', 'six', 'wide', 'outside', 'inside',
]);
/** The copy is gender-neutral: the classics include women's goals. */
const GENDERED = new Set(['he', 'his', 'him']);

/** The words in a line that give a position away or assume a gender. */
export function leaks(text: string): string[] {
  const words = text.toLowerCase().match(/[\p{L}']+/gu) ?? [];
  return words.filter((w) => BANNED.has(w) || GENDERED.has(w));
}

export function ordinal(n: number): string {
  const t = n % 100;
  if (t >= 11 && t <= 13) return `${n}th`;
  const end: Record<number, string> = { 1: 'st', 2: 'nd', 3: 'rd' };
  return `${n}${end[n % 10] ?? 'th'}`;
}

/** Understat's last action before the shot, where it says something. */
export const ACTIONS: Record<string, string> = {
  TakeOn: ' after a dribble',
  Dribble: ' after a dribble',
  Cross: ' from a cross',
  Throughball: ' from a through ball',
  Rebound: ' from a rebound',
  Chipped: ' from a chipped pass',
  HeadPass: ' from a flick-on',
  BallRecovery: ' after winning the ball back',
  Tackle: ' after winning the ball back',
  Interception: ' after an interception',
};

export interface ShotFacts {
  shotType: string;
  situation: string;
  lastAction: string;
  minute: number;
  /** Home–away after the goal, or null when the shots did not add up to the result. */
  score: [number, number] | null;
}

/** "A header from a corner, in the 42nd minute, to make it 3–1." Which foot is never said. */
export function recentLine(s: ShotFacts): string {
  const free = s.situation === 'DirectFreekick';
  const kind = free ? 'A direct free kick' : s.shotType === 'Head' ? 'A header' : 'A shot';
  const how = free ? ''
    : s.situation === 'FromCorner' ? ' from a corner'
    : s.situation === 'SetPiece' ? ' from a set piece'
    : ACTIONS[s.lastAction] ?? '';
  const score = s.score ? `, to make it ${s.score[0]}–${s.score[1]}` : '';
  return `${kind}${how}, in the ${ordinal(Math.max(1, s.minute))} minute${score}.`;
}
```

```ts
import { dailyOrder, dayNumber } from '../../../src/lib/daily';
import { inArea, type Spot } from './pitch';
import { par } from './score';
import type { Goals, Pack } from './types';

/** Day 1 of Pitch pin. Set to the launch date when it ships. */
export const PIN_EPOCH = '2026-10-08';
/** A board's par must be at least this: an average of 7 m a goal from the penalty spot. */
export const PAR_FLOOR = 42;

/** A real calendar date in YYYY-MM-DD. Also what keeps a board id from being a path. */
export function isDate(s: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(s)) return false;
  const t = Date.parse(`${s}T00:00:00Z`);
  return !Number.isNaN(t) && new Date(t).toISOString().slice(0, 10) === s;
}

/** A fair board: the lazy pin cannot coast it, and it has goals from in and out of the area. */
export function fair(ats: readonly Spot[]): boolean {
  return par(ats).metres >= PAR_FLOOR && ats.some(inArea) && ats.some((a) => !inArea(a));
}

/** The classics the daily draws from: those in no pack, so a pack never spoils a daily. */
export function dailyPool(goals: Goals, packs: readonly Pack[]): string[] {
  const inPack = new Set(packs.flatMap((p) => p.goals));
  return Object.keys(goals).filter((s) => goals[s]!.kind === 'classic' && !inPack.has(s)).sort();
}

/** The classics for a day: n at a time from a fixed shuffle, wrapping when the pool runs out. */
export function classicsFor(day: number, pool: readonly string[], n = 2): string[] {
  const order = dailyOrder(pool);
  const len = order.length;
  return Array.from({ length: n }, (_, i) => order[((((day - 1) * n + i) % len) + len) % len]!);
}

/** A day's six goals: the week script's board, or six classics when it was not run. */
export function daySlugs(date: string, days: Readonly<Record<string, string[]>>, pool: readonly string[]): string[] {
  return days[date] ?? classicsFor(dayNumber(date, PIN_EPOCH), pool, 6);
}
```

```ts
import { dayNumber } from '../../../src/lib/daily';
import { isDate, PIN_EPOCH } from './boards';
import type { Spot } from './pitch';
import type { Answer } from './types';

/**
 * What the player has done, kept in the browser. No accounts: a throw is
 * written when the answer comes back, and the first throw at a goal is the
 * one that counts. A refresh resumes from here.
 */
export interface Throw { pin: Spot; answer: Answer }
/** `finished` is the local date the last pin landed, or null. */
export interface PinPlay { throws: (Throw | null)[]; finished: string | null }
export type Store = Record<string, PinPlay>;

const KEY = 'pin.progress.v1';

export function readStore(): Store {
  try {
    const raw = localStorage.getItem(KEY);
    const parsed: unknown = raw ? JSON.parse(raw) : {};
    return parsed && typeof parsed === 'object' ? (parsed as Store) : {};
  } catch {
    return {};
  }
}

export function recordThrow(board: string, n: number, i: number, t: Throw, today: string): Store {
  const store = readStore();
  const old = store[board];
  const play: PinPlay = old && old.throws.length === n ? old : { throws: Array<Throw | null>(n).fill(null), finished: null };
  if (!play.throws[i]) play.throws[i] = t;
  if (play.finished === null && play.throws.every(Boolean)) play.finished = today;
  store[board] = play;
  try { localStorage.setItem(KEY, JSON.stringify(store)); } catch { /* private mode, or full */ }
  return store;
}

/** Consecutive dailies finished on their own day, ending today or yesterday. Packs do not count. */
export function streakOf(store: Store, today: string): number {
  const days = new Set<number>();
  for (const [id, p] of Object.entries(store)) if (isDate(id) && p.finished === id) days.add(dayNumber(id, PIN_EPOCH));
  const t = dayNumber(today, PIN_EPOCH);
  let d = days.has(t) ? t : t - 1;
  let n = 0;
  while (days.has(d)) { n++; d--; }
  return n;
}
```

```ts
import { fair } from './boards';
import { distance, SPOT, type Spot } from './pitch';

/** A recent goal the week script could use. */
export interface Candidate { slug: string; league: string; matchId: number; xg: number; at: Spot }

const LEAGUES = ['EPL', 'La_liga', 'Bundesliga', 'Serie_A', 'Ligue_1'];
/** A goal the lazy pin is this close to tests nothing. */
const TOO_EASY = 4;

/**
 * The memorable first: the Premier League before the rest, then the lowest
 * xG (a 0.04 goal is the one in the highlights), at most one goal a match,
 * and none the lazy pin would land on.
 */
export function rankRecent(cs: readonly Candidate[]): Candidate[] {
  const seen = new Set<number>();
  return cs
    .filter((c) => distance(SPOT, c.at) >= TOO_EASY)
    .sort((a, b) => LEAGUES.indexOf(a.league) - LEAGUES.indexOf(b.league) || a.xg - b.xg)
    .filter((c) => (seen.has(c.matchId) ? false : (seen.add(c.matchId), true)));
}

/**
 * Boards for the given dates: four recent goals, in rank order, and the
 * day's two classics, as recent, recent, classic, recent, recent, classic.
 * A board that would be unfair takes a later goal in place of one of the
 * four. Throws when the goals run out, so a short week is never written.
 */
export function planWeek(
  dates: readonly string[],
  ranked: readonly Candidate[],
  classicsOf: (date: string) => string[],
  atOf: (slug: string) => Spot,
): Record<string, string[]> {
  const queue = [...ranked];
  const out: Record<string, string[]> = {};
  for (const date of dates) {
    const classics = classicsOf(date);
    const ok = (picks: Candidate[]) => fair([...picks.map((p) => p.at), ...classics.map(atOf)]);
    if (queue.length < 4) throw new Error(`Not enough recent goals for ${date}`);
    let picks = queue.slice(0, 4);
    if (!ok(picks)) {
      const swapped = swapIn(queue.slice(4), picks, ok);
      if (!swapped) throw new Error(`No fair board for ${date}`);
      picks = swapped;
    }
    for (const p of picks) queue.splice(queue.indexOf(p), 1);
    const [r0, r1, r2, r3] = picks.map((p) => p.slug);
    out[date] = [r0!, r1!, classics[0]!, r2!, r3!, classics[1]!];
  }
  return out;
}

/** Try each later goal in each of the four places, last place first. */
function swapIn(later: Candidate[], picks: Candidate[], ok: (p: Candidate[]) => boolean): Candidate[] | null {
  for (const c of later) {
    for (let i = picks.length - 1; i >= 0; i--) {
      const next = picks.map((p, j) => (j === i ? c : p));
      if (ok(next)) return next;
    }
  }
  return null;
}
```

```ts
/**
 * Pitch pin's classics. For each hand-picked goal in
 * games/pin/data/classics.source.json, load its StatsBomb event file, find
 * the goal, and write its spot and xG into games/pin/data/goals.json.
 * Recent goals already in the file are kept.
 *
 *   npx tsx scripts/fetch-pin-classics.ts
 *
 * StatsBomb open data is for research, credited with their logo, and not to
 * be redistributed: only the chosen goals' spots are committed.
 */
import { readFile, writeFile } from 'node:fs/promises';
import { fromStatsBomb, round1 } from '../games/pin/lib/pitch';
import type { Goal, Goals } from '../games/pin/lib/types';

interface Source {
  slug: string; scorer: string; match: string; date: string; competition: string; line: string;
  matchId: number; player: string; minute: number;
}
interface SbEvent {
  id: string;
  type: { name: string };
  minute: number;
  player?: { name: string };
  location?: [number, number];
  shot?: { outcome: { name: string }; type: { name: string }; statsbomb_xg: number };
}

const BASE = 'https://raw.githubusercontent.com/statsbomb/open-data/master/data/events';
const GOALS = 'games/pin/data/goals.json';

const sources = JSON.parse(await readFile('games/pin/data/classics.source.json', 'utf8')) as Source[];
const existing = JSON.parse(await readFile(GOALS, 'utf8')) as Goals;
const out: Goals = Object.fromEntries(Object.entries(existing).filter(([, g]) => g.kind !== 'classic'));
const events = new Map<number, SbEvent[]>();
const problems: string[] = [];

for (const s of sources) {
  if (!events.has(s.matchId)) {
    const res = await fetch(`${BASE}/${s.matchId}.json`);
    if (!res.ok) throw new Error(`${s.slug}: events ${s.matchId} gave ${res.status}`);
    events.set(s.matchId, (await res.json()) as SbEvent[]);
  }
  const hits = events.get(s.matchId)!.filter((e) =>
    e.type.name === 'Shot' && e.shot?.outcome.name === 'Goal' && e.minute === s.minute && e.player?.name.includes(s.player));
  if (hits.length !== 1) { problems.push(`${s.slug}: ${hits.length} goals by "${s.player}" in minute ${s.minute}`); continue; }
  const e = hits[0]!;
  if (e.shot!.type.name === 'Penalty') { problems.push(`${s.slug}: a penalty`); continue; }
  const at = fromStatsBomb(e.location![0], e.location![1]);
  const goal: Goal = {
    scorer: s.scorer, match: s.match, date: s.date, competition: s.competition, line: s.line, kind: 'classic',
    source: { from: 'statsbomb', matchId: s.matchId, eventId: e.id },
    at: { x: round1(at.x), y: round1(at.y) },
    xg: Math.round(e.shot!.statsbomb_xg * 100) / 100,
  };
  out[s.slug] = goal;
  console.log(`${s.slug.padEnd(26)} x ${goal.at.x}  y ${goal.at.y}  xG ${goal.xg}`);
}

if (problems.length) {
  console.error(problems.join('\n'));
  process.exit(1);
}
await writeFile(GOALS, `${JSON.stringify(out, null, 2)}\n`);
console.log(`${sources.length} classics written to ${GOALS}`);
```

```ts
import goalsJson from './data/goals.json';
import packsJson from './data/packs.json';
import daysJson from './data/days.json';
import { dayNumber } from '../../src/lib/daily';
import { dailyPool, daySlugs, isDate, PIN_EPOCH } from './lib/boards';
import { distance, inHalf, mirrorNote, type Spot } from './lib/pitch';
import { par } from './lib/score';
import type { Answer, BoardView, Goals, Pack } from './lib/types';

/**
 * Pitch pin's ref. The coordinates stay here: the page gets the goals
 * without them, and a throw gets back the one answer it was thrown at.
 */
const goals = goalsJson as unknown as Goals;
const packs = packsJson as unknown as Pack[];
const days = daysJson as unknown as Record<string, string[]>;
const pool = dailyPool(goals, packs);

/** The latest playable day: tomorrow in UTC, so every time zone has its today. */
const latest = (now: Date): string => new Date(now.getTime() + 86400000).toISOString().slice(0, 10);

interface Slugs { slugs: string[]; title: string; date: string | null; number: number | null }

function slugsFor(id: string, now: Date): Slugs | null {
  if (id.startsWith('pack:')) {
    const p = packs.find((x) => x.id === id.slice(5));
    return p ? { slugs: p.goals, title: p.title, date: null, number: null } : null;
  }
  if (!isDate(id) || id < PIN_EPOCH || id > latest(now)) return null;
  const n = dayNumber(id, PIN_EPOCH);
  return { slugs: daySlugs(id, days, pool), title: `No. ${n}`, date: id, number: n };
}

export function boardView(id: string, now = new Date()): BoardView | null {
  const s = slugsFor(id, now);
  if (!s || s.slugs.some((slug) => !goals[slug])) return null;
  return {
    id,
    title: s.title,
    date: s.date,
    number: s.number,
    par: par(s.slugs.map((slug) => goals[slug]!.at)),
    goals: s.slugs.map((slug) => {
      const g = goals[slug]!;
      return { slug, scorer: g.scorer, match: g.match, date: g.date, competition: g.competition, line: g.line, credit: g.source.from };
    }),
  };
}

export function packList(): { id: string; title: string }[] {
  return packs.map(({ id, title }) => ({ id, title }));
}

export interface ThrowRequest { board: string; goal: string; pin: Spot }

/** A throw as the route accepts it: short strings and a finite point inside the half. */
export function parseThrow(body: unknown): ThrowRequest | null {
  if (!body || typeof body !== 'object' || Array.isArray(body)) return null;
  const b = body as Record<string, unknown>;
  if (typeof b.board !== 'string' || typeof b.goal !== 'string' || b.board.length > 40 || b.goal.length > 60) return null;
  if (typeof b.x !== 'number' || typeof b.y !== 'number' || !Number.isFinite(b.x) || !Number.isFinite(b.y)) return null;
  const pin = { x: b.x, y: b.y };
  return inHalf(pin) ? { board: b.board, goal: b.goal, pin } : null;
}

/** The answer to one throw, or null when that goal is not on a playable board. */
export function judge({ board, goal, pin }: ThrowRequest, now = new Date()): Answer | null {
  const s = slugsFor(board, now);
  const g = goals[goal];
  if (!s || !g || !s.slugs.includes(goal)) return null;
  return { at: g.at, distance: distance(pin, g.at), mirrored: mirrorNote(pin, g.at), xg: g.xg, credit: g.source.from };
}
```

```ts
import { judge, parseThrow } from '../../../../games/pin/server';

/** A thrown pin, judged. The answer for that one goal, never the rest of the board. */
export async function POST(req: Request) {
  const t = parseThrow(await req.json().catch(() => null));
  if (!t) return Response.json({ error: 'bad request' }, { status: 400 });
  const answer = judge(t);
  if (!answer) return Response.json({ error: 'not found' }, { status: 404 });
  return Response.json(answer);
}
```

```ts
 *   npx tsx scripts/build-pin-week.ts [--from YYYY-MM-DD] [--days 21]
 *
 * Looks back --days for results in five leagues. Understat publishes no
 * terms: it is called one request at a time with a pause between, and only
 * the chosen goals are kept. Run by hand, once a week; nothing schedules it.
 */
import { readFile, writeFile } from 'node:fs/promises';
import { dateOfDay, dayNumber, localDate } from '../src/lib/daily';
import { classicsFor, dailyPool, PIN_EPOCH } from '../games/pin/lib/boards';
import { recentLine } from '../games/pin/lib/line';
import { fromUnderstat, round1 } from '../games/pin/lib/pitch';
import { planWeek, rankRecent, type Candidate } from '../games/pin/lib/week';
import type { Goal, Goals, Pack } from '../games/pin/lib/types';

const LEAGUES: Record<string, string> = {
  EPL: 'Premier League', La_liga: 'La Liga', Bundesliga: 'Bundesliga', Serie_A: 'Serie A', Ligue_1: 'Ligue 1',
};
const HEADERS = { 'X-Requested-With': 'XMLHttpRequest', 'User-Agent': 'Mozilla/5.0' };

interface Fixture { id: string; isResult: boolean; datetime: string; goals: { h: string; a: string } }
interface Shot {
  id: string; minute: string; result: string; X: string; Y: string; xG: string; player: string; h_a: 'h' | 'a';
  situation: string; shotType: string; lastAction: string; h_team: string; a_team: string; date: string;
}

const arg = (name: string): string | undefined => {
  const i = process.argv.indexOf(`--${name}`);
  return i > 0 ? process.argv[i + 1] : undefined;
};
const pause = (ms: number) => new Promise((r) => setTimeout(r, ms));

/** Understat is slow and sometimes refuses: time out, back off, retry. */
async function get<T>(url: string): Promise<T> {
  for (let i = 0; i < 4; i++) {
    try {
      const res = await fetch(url, { headers: HEADERS, signal: AbortSignal.timeout(30_000) });
      if (res.ok) return (await res.json()) as T;
    } catch { /* slow or refused */ }
    await pause(3000 * (i + 1));
  }
  throw new Error(`Understat did not answer: ${url}`);
}

const from = arg('from') ?? localDate();
const windowDays = Number(arg('days') ?? 21);
const now = new Date();
const season = now.getUTCMonth() >= 6 ? now.getUTCFullYear() : now.getUTCFullYear() - 1;
const since = new Date(now.getTime() - windowDays * 86400000).toISOString().slice(0, 10);

const goals = JSON.parse(await readFile('games/pin/data/goals.json', 'utf8')) as Goals;
const packs = JSON.parse(await readFile('games/pin/data/packs.json', 'utf8')) as Pack[];
const days = JSON.parse(await readFile('games/pin/data/days.json', 'utf8')) as Record<string, string[]>;

const found: Record<string, Goal> = {};
const candidates: Candidate[] = [];

for (const [league, name] of Object.entries(LEAGUES)) {
  const data = await get<{ dates: Fixture[] }>(`https://understat.com/getLeagueData/${league}/${season}`);
  const played = data.dates.filter((f) => f.isResult && f.datetime.slice(0, 10) >= since);
  console.log(`${name}: ${played.length} results since ${since}`);
  for (const f of played) {
    await pause(1000);
    const m = await get<{ shots: { h: Shot[]; a: Shot[] } }>(`https://understat.com/getMatchData/${f.id}`);
    const shots = [...m.shots.h, ...m.shots.a].sort((p, q) => Number(p.minute) - Number(q.minute) || Number(p.id) - Number(q.id));
    // The score after each goal. An own goal is filed under the side of the
    // player who scored it and counts for the other side (checked on 50 matches).
    const score = { h: 0, a: 0 };
    const after = new Map<string, [number, number]>();
    for (const s of shots) {
      if (s.result === 'Goal') score[s.h_a]++;
      else if (s.result === 'OwnGoal') score[s.h_a === 'h' ? 'a' : 'h']++;
      else continue;
      after.set(s.id, [score.h, score.a]);
    }
    const trusted = score.h === Number(f.goals.h) && score.a === Number(f.goals.a);
    for (const s of shots) {
      if (s.result !== 'Goal' || s.situation === 'Penalty') continue;
      const slug = `u-${s.id}`;
      if (goals[slug]) continue; // already on a board
      const raw = fromUnderstat(Number(s.X), Number(s.Y));
      const at = { x: round1(raw.x), y: round1(raw.y) };
      const xg = Math.round(Number(s.xG) * 100) / 100;
      found[slug] = {
        scorer: s.player,
        match: `${s.h_team} v ${s.a_team}`,
        date: s.date.slice(0, 10),
        competition: `${name} ${season}/${String((season + 1) % 100).padStart(2, '0')}`,
        line: recentLine({ shotType: s.shotType, situation: s.situation, lastAction: s.lastAction, minute: Number(s.minute), score: trusted ? after.get(s.id)! : null }),
        kind: 'recent',
        source: { from: 'understat', matchId: Number(f.id), shotId: s.id },
        at,
        xg,
      };
      candidates.push({ slug, league, matchId: Number(f.id), xg, at });
    }
  }
}

// The next seven days that have no board yet.
const dates: string[] = [];
for (let d = dayNumber(from, PIN_EPOCH); dates.length < 7; d++) {
  const date = dateOfDay(d, PIN_EPOCH);
  if (!days[date]) dates.push(date);
}

const pool = dailyPool(goals, packs);
const week = planWeek(
  dates,
  rankRecent(candidates),
  (date) => classicsFor(dayNumber(date, PIN_EPOCH), pool),
  (slug) => (found[slug] ?? goals[slug])!.at,
);
for (const [date, slugs] of Object.entries(week)) {
  days[date] = slugs;
  for (const s of slugs) if (found[s]) goals[s] = found[s]!;
  console.log(date, slugs.map((s) => `${goals[s]!.scorer} (${goals[s]!.kind})`).join(', '));
}
await writeFile('games/pin/data/goals.json', `${JSON.stringify(goals, null, 2)}\n`);
await writeFile('games/pin/data/days.json', `${JSON.stringify(Object.fromEntries(Object.entries(days).sort()), null, 2)}\n`);
console.log(`${Object.keys(week).length} boards written`);
```

```tsx
'use client';
import { useEffect, useId, useRef, useState, type KeyboardEvent, type PointerEvent } from 'react';
import { clamp, distance, type Spot } from '../lib/pitch';

/**
 * The pitch, in metres: one SVG user unit is one metre, goal at the top.
 * Three uses: aiming (a pin to place), the reveal (zoom onto the pin and
 * the true spot, rings, the line) and full time (all six pairs).
 */
interface Box { x: number; y: number; w: number; h: number }
interface Pair { pin: Spot; at: Spot }

/** The whole half, with room for the goal above the line. */
const FULL: Box = { x: -1.5, y: -3, w: 71, h: 57 };
/** While a thumb drags, the pin sits this far above it, in screen pixels. */
const LIFT = 48;
const ZOOM_MS = 420;
const HOLD_MS = 1300;
const ARROWS: Record<string, [number, number]> = { ArrowLeft: [-1, 0], ArrowRight: [1, 0], ArrowUp: [0, -1], ArrowDown: [0, 1] };

const viewBox = (b: Box) => `${b.x} ${b.y} ${b.w} ${b.h}`;
const ease = (t: number) => 1 - (1 - t) ** 3;
const lerp = (a: Box, b: Box, t: number): Box => ({ x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t, w: a.w + (b.w - a.w) * t, h: a.h + (b.h - a.h) * t });

/** A box round the pin and the spot, in the pitch's proportions. */
function focusOn({ pin, at }: Pair): Box {
  const h = Math.max(18, distance(pin, at) + 10);
  const w = (h * FULL.w) / FULL.h;
  return { x: (pin.x + at.x) / 2 - w / 2, y: (pin.y + at.y) / 2 - h / 2, w, h };
}

interface Props {
  label: string;
  /** Aiming: the pin so far, and what to do when it moves. */
  pin?: Spot | null;
  onPin?: (s: Spot) => void;
  onEnter?: () => void;
  /** The reveal of one throw. */
  reveal?: Pair | null;
  /** Full time: every throw. */
  all?: Pair[];
}

export default function Pitch({ label, pin = null, onPin, onEnter, reveal = null, all }: Props) {
  const svg = useRef<SVGSVGElement>(null);
  const uid = useId().replace(/[^a-zA-Z0-9_-]/g, '');
  const [box, setBox] = useState<Box>(FULL);
  const [drag, setDrag] = useState<'touch' | 'mouse' | null>(null);

  const rx = reveal?.pin.x, ry = reveal?.pin.y, ax = reveal?.at.x, ay = reveal?.at.y;
  useEffect(() => {
    setBox(FULL);
    if (rx === undefined || ry === undefined || ax === undefined || ay === undefined) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const near = focusOn({ pin: { x: rx, y: ry }, at: { x: ax, y: ay } });
    const t0 = performance.now();
    let raf = 0;
    const step = (now: number) => {
      const t = now - t0;
      if (t < ZOOM_MS) setBox(lerp(FULL, near, ease(t / ZOOM_MS)));
      else if (t < ZOOM_MS + HOLD_MS) setBox(near);
      else if (t < 2 * ZOOM_MS + HOLD_MS) setBox(lerp(near, FULL, ease((t - ZOOM_MS - HOLD_MS) / ZOOM_MS)));
      else { setBox(FULL); return; }
      raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
  }, [rx, ry, ax, ay]);

  function toSpot(e: PointerEvent<SVGSVGElement>, kind: 'touch' | 'mouse'): Spot | null {
    const ctm = svg.current?.getScreenCTM();
    if (!ctm) return null;
    const p = new DOMPoint(e.clientX, e.clientY - (kind === 'touch' ? LIFT : 0)).matrixTransform(ctm.inverse());
    return clamp({ x: p.x, y: p.y });
  }
  function down(e: PointerEvent<SVGSVGElement>) {
    if (!onPin) return;
    const kind = e.pointerType === 'mouse' ? 'mouse' : 'touch';
    e.currentTarget.setPointerCapture(e.pointerId);
    setDrag(kind);
    const s = toSpot(e, kind);
    if (s) onPin(s);
  }
  function move(e: PointerEvent<SVGSVGElement>) {
    if (!onPin || !drag) return;
    const s = toSpot(e, drag);
    if (s) onPin(s);
  }
  function key(e: KeyboardEvent<SVGSVGElement>) {
    if (!onPin) return;
    if (e.key === 'Enter') { e.preventDefault(); onEnter?.(); return; }
    const d = ARROWS[e.key];
    if (!d) return;
    e.preventDefault();
    const step = e.shiftKey ? 2 : 0.5;
    const from = pin ?? { x: 34, y: 20 };
    onPin(clamp({ x: from.x + d[0] * step, y: from.y + d[1] * step }));
  }

  const lines = `${uid}-lines`;
  const half = `${uid}-half`;
  const magClip = `${uid}-mag`;
  // The magnifier (about 96 px across at phone width) sits in the top corner
  // away from the pin, so neither the thumb nor the lens covers it.
  const mag = drag === 'touch' && pin ? { x: pin.x < 34 ? 58 : 10, y: 9, r: 9 } : null;

  return (
    <div className="pitch">
      <svg
        ref={svg}
        className={`pitch__svg${onPin ? ' pitch__svg--aim' : ''}`}
        viewBox={viewBox(box)}
        role="img"
        aria-label={label}
        tabIndex={onPin ? 0 : undefined}
        onPointerDown={down}
        onPointerMove={move}
        onPointerUp={() => setDrag(null)}
        onPointerCancel={() => setDrag(null)}
        onKeyDown={key}
      >
        <defs>
          <g id={lines}><Lines /></g>
          <clipPath id={half}><rect x={0} y={0} width={68} height={52.5} /></clipPath>
          {mag ? <clipPath id={magClip}><circle cx={mag.x} cy={mag.y} r={mag.r} /></clipPath> : null}
        </defs>
        <rect className="pitch__grass" x={FULL.x} y={FULL.y} width={FULL.w} height={FULL.h} />
        <use href={`#${lines}`} />

        {reveal ? <Rings at={reveal.at} clip={half} /> : null}
        {reveal ? <line className="pitch__trail" x1={reveal.pin.x} y1={reveal.pin.y} x2={reveal.at.x} y2={reveal.at.y} /> : null}
        {reveal ? <PinMark at={reveal.pin} /> : null}
        {reveal ? <Ball at={reveal.at} /> : null}

        {all?.map((p, i) => (
          <g key={i}>
            <line className="pitch__trail pitch__trail--still" x1={p.pin.x} y1={p.pin.y} x2={p.at.x} y2={p.at.y} />
            <Ball at={p.at} still />
            <PinMark at={p.pin} n={i + 1} />
          </g>
        ))}

        {pin && !reveal ? <PinMark at={pin} /> : null}

        {mag && pin ? (
          <g className="pitch__mag" aria-hidden>
            <g clipPath={`url(#${magClip})`}>
              <circle className="pitch__magground" cx={mag.x} cy={mag.y} r={mag.r} />
              <g transform={`translate(${mag.x} ${mag.y}) scale(2.5) translate(${-pin.x} ${-pin.y})`}>
                <use href={`#${lines}`} />
                <PinMark at={pin} />
              </g>
            </g>
            <circle className="pitch__magring" cx={mag.x} cy={mag.y} r={mag.r} />
          </g>
        ) : null}
      </svg>
    </div>
  );
}
```

```tsx
'use client';
import { useEffect, useState } from 'react';
import Pitch from './Pitch';
import FullTime from './FullTime';
import { dateOfDay, dayNumber, localDate, longDate } from '../../../src/lib/daily';
import { PIN_EPOCH } from '../lib/boards';
import { readStore, recordThrow, streakOf, type Throw } from '../lib/progress';
import { points, ring, RING_NAME, tally, type Mode } from '../lib/score';
import type { Spot } from '../lib/pitch';
import type { Answer, BoardView, GoalCard as Card } from '../lib/types';

/**
 * A Pitch pin board: six goals, one throw each. The page holds no answers;
 * each throw goes to /api/pin/throw and the answer comes back for that goal
 * alone. Progress is kept in the browser so a refresh resumes.
 */
export default function PitchPin({ board, mode, packs }: { board: BoardView; mode: Mode; packs: { id: string; title: string }[] }) {
  const n = board.goals.length;
  const [throws, setThrows] = useState<(Throw | null)[]>(() => Array<Throw | null>(n).fill(null));
  const [loaded, setLoaded] = useState(false);
  const [pin, setPin] = useState<Spot | null>(null);
  const [viewing, setViewing] = useState<number | null>(null);
  const [busy, setBusy] = useState(false);
  const [failed, setFailed] = useState(false);
  const [streak, setStreak] = useState(0);

  useEffect(() => {
    const store = readStore();
    const saved = store[board.id]?.throws;
    if (saved && saved.length === n) setThrows(saved);
    setStreak(streakOf(store, localDate()));
    setLoaded(true);
  }, [board.id, n]);

  const next = throws.findIndex((t) => !t);
  const done = next === -1 && viewing === null;
  const current = viewing ?? Math.max(next, 0);
  const goal = board.goals[current]!;
  const query = mode === 'points' ? '?score=points' : '';

  async function throwPin() {
    if (!pin || busy || next === -1) return;
    setBusy(true);
    setFailed(false);
    try {
      const res = await fetch('/api/pin/throw', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ board: board.id, goal: board.goals[next]!.slug, x: pin.x, y: pin.y }),
      });
      if (!res.ok) throw new Error(String(res.status));
      const answer = (await res.json()) as Answer;
      const store = recordThrow(board.id, n, next, { pin, answer }, localDate());
      setThrows(store[board.id]!.throws);
      setStreak(streakOf(store, localDate()));
      setViewing(next);
      setPin(null);
    } catch {
      setFailed(true);
    } finally {
      setBusy(false);
    }
  }

  const shown = viewing !== null ? throws[viewing] : null;
  const credits = [...new Set(board.goals.map((g) => g.credit))]
    .map((c) => (c === 'statsbomb' ? 'StatsBomb open data' : 'Understat'))
    .join(' and ');
  const pointsSoFar = tally(throws.filter((t): t is Throw => !!t).map((t) => t.answer.distance)).points;

  return (
    <div className="app pin">
      <header className="masthead">
        <a className="masthead__brand display" href={`/pin${query}`}>
          <PinLogo />
          Pitch pin
        </a>
        <div className="masthead__meta"><span>{board.title}</span></div>
      </header>

      <main className="pin__main">
        {!loaded ? null : done ? (
          <FullTime board={board} throws={throws as Throw[]} mode={mode} streak={board.date ? streak : null} />
        ) : (
          <>
            <Ticks n={n} thrown={throws.filter(Boolean).length} current={current} />
            <GoalCard goal={goal} small={shown !== null} />
            {shown ? (
              <>
                <Pitch label={`Your pin, ${shown.answer.distance.toFixed(1)} metres from where it was struck`} reveal={{ pin: shown.pin, at: shown.answer.at }} />
                <Result
                  t={shown}
                  goal={goal}
                  mode={mode}
                  pointsSoFar={pointsSoFar}
                  last={next === -1}
                  onNext={() => setViewing(null)}
                />
              </>
            ) : (
              <>
                <Pitch label="The attacking half of a pitch, goal at the top. Place your pin." pin={pin} onPin={setPin} onEnter={throwPin} />
                <p className="pin__hint">Where was it struck from? Tap the pitch, drag to adjust.</p>
                {failed ? <p className="pin__error" role="alert">Could not reach the ref. Try again.</p> : null}
                <button type="button" className="pin__throw display" disabled={!pin || busy} onClick={throwPin}>Throw</button>
              </>
            )}
          </>
        )}
        <More board={board} packs={packs} query={query} />
        <p className="pin__credit">Shot data: {credits}. Metres on a standard pitch.</p>
      </main>
    </div>
  );
}

function PinLogo() {
  return (
    <svg className="masthead__mark masthead__mark--pin" viewBox="0 0 22 30" aria-hidden>
      <circle cx="11" cy="10" r="8.5" fill="none" stroke="currentColor" strokeWidth="2.6" />
      <circle cx="11" cy="10" r="3.2" fill="currentColor" />

function Ticks({ n, thrown, current }: { n: number; thrown: number; current: number }) {

function GoalCard({ goal, small }: { goal: Card; small: boolean }) {
  return (
    <section className={`pin__card${small ? ' pin__card--small' : ''}`}>
      {small ? null : <span className="pin__comp">{goal.competition}</span>}
      <h1 className="pin__scorer display">{goal.scorer}</h1>
      <span className="pin__match">{goal.match}, {longDate(goal.date)}</span>
      {small ? null : <p className="pin__line">{goal.line}</p>}
    </section>

const xgText = (a: Answer) =>
  `${a.credit === 'statsbomb' ? 'StatsBomb' : 'Understat'} gave it ${a.xg < 0.01 ? 'under a 1%' : `a ${Math.round(a.xg * 100)}%`} chance.`;

const watchUrl = (g: Card) =>
  `https://www.youtube.com/results?search_query=${encodeURIComponent(`${g.scorer} goal ${g.match} ${g.date.slice(0, 4)}`)}`;

function Result({ t, goal, mode, pointsSoFar, last, onNext }: {
  t: Throw; goal: Card; mode: Mode; pointsSoFar: number; last: boolean; onNext: () => void;
}) {
  const d = t.answer.distance;
  const r = ring(d);
  return (
    <section className="pin__result" aria-live="polite">
      <div className="pin__score">
        {mode === 'points' ? (
          <>
            <span className="pin__big display">+{points(d)}</span>
            <span className="pin__sofar"><span>So far</span><b className="display">{pointsSoFar}</b></span>
          </>
        ) : (
          <>
            <span className="pin__big display">{d.toFixed(1)} m</span>
            <span className={`pin__ring pin__ring--${r} display`}>{RING_NAME[r]}</span>
          </>
        )}
      </div>
      <p className="pin__fact">
        {mode === 'points' ? `${d.toFixed(1)} m off. ` : ''}Struck {t.answer.at.y.toFixed(1)} m out. {xgText(t.answer)}
      </p>
      {t.answer.mirrored !== null ? (
        <p className="pin__mirror">Wrong side? Mirrored, you were {t.answer.mirrored.toFixed(1)} m off.</p>
      ) : null}
      <div className="pin__actions">
        <a className="pin__watch" href={watchUrl(goal)} target="_blank" rel="noreferrer">Watch it</a>
        <button type="button" className="pin__next display" onClick={onNext}>{last ? 'Full time' : 'Next goal'}</button>
      </div>
    </section>
  );
}

/** Earlier days (up to a week) and the packs. Worked out from the player's own clock. */
function More({ board, packs, query }: { board: BoardView; packs: { id: string; title: string }[]; query: string }) {
  const [earlier, setEarlier] = useState<string[]>([]);
  useEffect(() => {
    const today = dayNumber(localDate(), PIN_EPOCH);
    const out: string[] = [];
    for (let d = today - 1; d >= 1 && out.length < 7; d--) out.push(dateOfDay(d, PIN_EPOCH));
    setEarlier(out);
  }, []);
  return (
    <nav className="pin__more" aria-label="More boards">
      {earlier.length ? (
        <div>
          <h2 className="pin__morehead display">Earlier days</h2>
          <ul>
            {earlier.map((d) => (
              <li key={d}><a href={`/pin/${d}${query}`} aria-current={d === board.id ? 'page' : undefined}>No. {dayNumber(d, PIN_EPOCH)}, {longDate(d)}</a></li>
            ))}
          </ul>
        </div>
      ) : null}
      <div>
        <h2 className="pin__morehead display">Packs</h2>
        <ul>
          {packs.map((p) => (
            <li key={p.id}><a href={`/pin/pack/${p.id}${query}`} aria-current={`pack:${p.id}` === board.id ? 'page' : undefined}>{p.title}</a></li>
          ))}
        </ul>
      </div>
    </nav>
  );
}
```

```tsx
/** Share where the phone can, else copy. A cancelled share is not a failure. */
function ShareButton({ text }: { text: string }) {
  const [state, setState] = useState<'idle' | 'copied' | 'failed'>('idle');
  async function share() {
    try {
      if (navigator.share) {
        await navigator.share({ text });
        return;
      }
      await navigator.clipboard.writeText(text);
      setState('copied');
    } catch (e) {
      if (e instanceof Error && e.name === 'AbortError') return;
      setState('failed');
    }
    setTimeout(() => setState('idle'), 2200);
  }
  return (
    <button type="button" className="pin__throw display" onClick={share}>
      {state === 'copied' ? 'Copied' : state === 'failed' ? 'Could not share' : 'Share'}
    </button>
  );
}
```
